# Kits Madness, étape 1 : format, kits locaux et traduction BMad — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Une équipe déclare ses références (charte, conventions, glossaire, règles) dans un kit au format `madness-kit.toml`, l'active sur un projet, et les agents BMad concernés les reçoivent par les surcouches natives `_bmad/custom/<compétence>.toml`.

**Architecture:** Le cœur pur (`src/core/kits/`) lit et valide un manifeste, puis calcule pour chaque compétence BMad les entrées à injecter. Les adaptateurs (`src/adapters/kits/`) gèrent les sources locales, la copie dans le projet, le verrou et l'écriture de blocs balisés dans les surcouches. Main expose un canal IPC `madness:kits` ; le rendu ajoute une section « Kits » aux réglages et un panneau « Kits du projet » dans le Parcours. Spec : `docs/superpowers/specs/2026-11-11-kits-madness-design.md` (étape 1 du § 11).

**Tech Stack:** TypeScript (Electron main + React 19), `smol-toml@1.9.1` pour lire le TOML, `node --test` pour les tests, Python embarqué + `uv` pour le test de bout en bout avec `resolve_customization.py`.

**Hors étape 1 :** sources git, projets de démarrage et de référence, compétences et MCP de kit, relecture de conformité par l'orchestrateur, maquettes. Le manifeste les accepte déjà (validation), mais Madness les ignore à l'activation en le disant.

export const KIT_ROLES = ['analyst', 'pm', 'ux', 'architect', 'po', 'dev', 'review', 'all'] as const
export type KitRole = typeof KIT_ROLES[number]
const SKILLS: Record<Exclude<KitRole, 'all'>, readonly string[]> = {
  analyst: ['bmad-agent-analyst', 'bmad-product-brief', 'bmad-prfaq', 'bmad-brainstorming', 'bmad-forge-idea', 'bmad-deep-recon'],

  po: ['bmad-agent-po'],
  dev: ['bmad-agent-dev', 'bmad-build', 'bmad-build-auto', 'bmad-qa-generate-e2e-tests'],
  review: ['bmad-code-review', 'bmad-review'],
}
export function skillsFor(roles: readonly KitRole[]): string[] {

import { parse } from 'smol-toml'
import { KIT_ROLES } from './roles.ts'
import type { KitRole } from './roles.ts'

export type KitLevel = 'impose' | 'consulte'
export type PieceKind = 'docs' | 'glossary' | 'rules' | 'skill' | 'starter' | 'reference' | 'mcp' | 'assets'
export type KitPiece = { kind: PieceKind; title: string; path: string | null; level: KitLevel; for: KitRole[]; source: string | null; name: string | null }
export type KitRule = { id: string; text: string; level: KitLevel; for: KitRole[] }
export type Kit = { id: string; name: string; version: string; description: string; pieces: KitPiece[]; rules: KitRule[] }
export type ManifestResult = { ok: true; kit: Kit; notices: string[] } | { ok: false; errors: string[] }

const KINDS: readonly PieceKind[] = ['docs', 'glossary', 'rules', 'skill', 'starter', 'reference', 'mcp', 'assets']
const LATER: readonly PieceKind[] = ['skill', 'starter', 'reference', 'mcp']
const text = (value: unknown): string => typeof value === 'string' ? value.trim() : ''
const insideKit = (path: string): boolean => !!path && !/^([a-z]:|[\\/])/i.test(path) && !path.split(/[\\/]/).includes('..')

function roles(value: unknown, where: string, errors: string[]): KitRole[] {
  const list = Array.isArray(value) ? value.map(text) : []
  if (!list.length) { errors.push(where + ' : « for » doit nommer au moins un rôle.'); return [] }
  for (const role of list) if (!(KIT_ROLES as readonly string[]).includes(role)) errors.push(where + ' : rôle « ' + role + ' » inconnu (' + KIT_ROLES.join(', ') + ').')
  return list.filter((role): role is KitRole => (KIT_ROLES as readonly string[]).includes(role))
}
const level = (value: unknown): KitLevel => value === 'impose' ? 'impose' : 'consulte'

export function parseKitManifest(source: string): ManifestResult {
  let data: Record<string, unknown>
  try { data = parse(source) as Record<string, unknown> } catch (error) { return { ok: false, errors: ['Manifeste illisible : ' + (error instanceof Error ? error.message.split('\n')[0] : 'TOML invalide')] } }
  const errors: string[] = [], notices: string[] = []
  const head = (data.kit && typeof data.kit === 'object' ? data.kit : {}) as Record<string, unknown>
  const id = text(head.id), name = text(head.name), version = text(head.version)
  if (!/^[a-z0-9][a-z0-9-]{0,62}$/.test(id)) errors.push('kit.id doit être en minuscules, chiffres et tirets.')
  if (!name) errors.push('kit.name est obligatoire.')
  if (!/^\d+\.\d+\.\d+(-[0-9a-z.-]+)?$/i.test(version)) errors.push('kit.version doit être une version sémantique (1.4.0).')
  if (head.format !== undefined && head.format !== 1) errors.push('format ' + String(head.format) + ' inconnu : cette version de Madness lit le format 1.')
  const pieces: KitPiece[] = (Array.isArray(data.piece) ? data.piece : []).map((raw: unknown, index: number) => {
    const piece = (raw ?? {}) as Record<string, unknown>, where = 'pièce ' + (index + 1), kind = text(piece.kind) as PieceKind
    if (!KINDS.includes(kind)) errors.push(where + ' : type « ' + kind + ' » inconnu (' + KINDS.join(', ') + ').')
    const path = text(piece.path) || null
    if (path && !insideKit(path)) errors.push(where + ' : le chemin « ' + path + ' » sort du kit.')
    if (!path && ['docs', 'glossary', 'rules', 'assets', 'skill'].includes(kind)) errors.push(where + ' : « path » est obligatoire pour une pièce ' + kind + '.')
    const result: KitPiece = { kind, title: text(piece.title) || text(piece.name) || path || kind, path, level: level(piece.level), for: kind === 'starter' ? [] : roles(piece.for, where, errors), source: text(piece.source) || null, name: text(piece.name) || null }
    if (LATER.includes(kind)) notices.push('La pièce « ' + result.title + ' » (' + kind + ') sera prise en charge dans une prochaine version de Madness.')
    return result
  })
  const rules: KitRule[] = (Array.isArray(data.rule) ? data.rule : []).map((raw: unknown, index: number) => {
    const rule = (raw ?? {}) as Record<string, unknown>, where = 'règle ' + (index + 1)
    if (!text(rule.text)) errors.push(where + ' : « text » est obligatoire.')
    return { id: text(rule.id) || 'regle-' + (index + 1), text: text(rule.text), level: level(rule.level), for: roles(rule.for, where, errors) }
  })
  if (errors.length) return { ok: false, errors }
  return { ok: true, kit: { id, name, version, description: text(head.description), pieces, rules }, notices }
}
```

import type { Kit } from './manifest.ts'
import { skillsFor } from './roles.ts'

export type OverrideEntries = { table: 'agent' | 'workflow'; persistent_facts: string[]; principles: string[]; review_guidance: string[] }
const TEXT_KINDS = ['docs', 'glossary', 'rules', 'assets']

/** Active kits → what each installed BMad skill must receive. `tables` maps installed skills to their customize table. */
export function translateKits(kits: readonly Kit[], tables: Record<string, 'agent' | 'workflow'>): Record<string, OverrideEntries> {
  const out: Record<string, OverrideEntries> = {}
  const entry = (skill: string): OverrideEntries | null => {
    const table = tables[skill]; if (!table) return null
    return out[skill] ??= { table, persistent_facts: [], principles: [], review_guidance: [] }
  }
  for (const kit of kits) {
    for (const piece of kit.pieces) {
      if (!TEXT_KINDS.includes(piece.kind) || !piece.path) continue
      const fact = 'file:{project-root}/_bmad/kits/' + kit.id + '/' + piece.path
      for (const skill of skillsFor(piece.for)) {
        const target = entry(skill); if (!target) continue
        target.persistent_facts.push(fact)
        if (piece.level !== 'impose') continue
        if (target.table === 'agent') target.principles.push('Respecter obligatoirement « ' + piece.title + ' » du kit ' + kit.name + '.')
        if (skill === 'bmad-review') target.review_guidance.push(fact)
      }
    }
    for (const rule of kit.rules) for (const skill of skillsFor(rule.for)) {
      const target = entry(skill); if (!target) continue
      target.persistent_facts.push((rule.level === 'impose' ? 'OBLIGATOIRE' : 'Recommandé') + ' (kit ' + kit.name + ') : ' + rule.text)
      if (rule.level === 'impose' && target.table === 'agent') target.principles.push(rule.text)
      if (rule.level === 'impose' && skill === 'bmad-review') target.review_guidance.push(rule.text)
    }
  }
  return out
}
```

### Task 5: Blocs balisés dans les surcouches BMad

**Files:** Create: `src/adapters/kits/overrides.ts`; Test: `tests/kits-overrides.test.mjs`

BMad ne lit que deux surcouches (`<compétence>.toml`, `<compétence>.user.toml`) et TOML interdit de redéclarer une table. Stratégie :
- fichier absent ou sans la table → bloc complet ajouté en fin de fichier : `# madness:kits:begin` / `[agent]` / clés / `# madness:kits:end` ;
- table présente sans les clés → bloc inséré juste après l'en-tête de table (clés seules) ;
- clé déjà présente, écrite à la main → refus explicite (`KIT_OVERRIDE_CONFLICT`), avec le nom du fichier et de la clé. Madness n'écrase jamais un réglage manuel.
Le retrait supprime exactement le bloc. Le résultat est relu avec `smol-toml` avant écriture.

```ts
import { parse } from 'smol-toml'
import type { OverrideEntries } from '../../core/kits/translate.ts'

const BEGIN = '# madness:kits:begin — écrit par Madness, ne pas modifier', END = '# madness:kits:end'
const KEYS = ['persistent_facts', 'principles', 'review_guidance'] as const
const quote = (value: string): string => JSON.stringify(value)

export function removeManagedBlock(text: string): string {
  const start = text.indexOf(BEGIN); if (start < 0) return text
  const end = text.indexOf(END, start); if (end < 0) throw Error('KIT_OVERRIDE_DAMAGED')
  let after = end + END.length; if (text[after] === '\n') after++
  return text.slice(0, start) + text.slice(after)
}

/** Writes Madness's entries into a BMad override, next to whatever the team wrote by hand. */
export function applyManagedBlock(original: string, entries: OverrideEntries): string {
  const base = removeManagedBlock(original)
  const keys = KEYS.filter(key => entries[key].length)
  if (!keys.length) return base
  const existing = (base.trim() ? parse(base) : {}) as Record<string, Record<string, unknown> | undefined>
  const table = existing[entries.table]
  for (const key of keys) if (table && key in table) throw Error('KIT_OVERRIDE_CONFLICT: ' + key)
  const lines = keys.map(key => key + ' = [\n' + entries[key].map(value => '  ' + quote(value) + ',').join('\n') + '\n]')
  let text: string
  if (!table) text = base + (base && !base.endsWith('\n') ? '\n' : '') + BEGIN + '\n[' + entries.table + ']\n' + lines.join('\n') + '\n' + END + '\n'
  else {
    const header = new RegExp('^\\[' + entries.table + '\\][ \\t]*(#.*)?$', 'm').exec(base)!
    const at = header.index + header[0].length + 1
    text = base.slice(0, at) + BEGIN + '\n' + lines.join('\n') + '\n' + END + '\n' + base.slice(at)
  }
  parse(text)
  return text
}
```

/** Rebuilds every override from the kits recorded in the lock, so files never drift from it. */
async function rebuild(root: string): Promise<string[]> {
  const kits = await Promise.all((await activeKits(root)).map(async k => (await readKit(kitDir(root, k.id))).kit))
  const tables = await skillTables(root), wanted = translateKits(kits, tables), custom = join(root, '_bmad', 'custom')

  }

}

export async function activateKit(root: string, folder: string): Promise<{ kit: Kit; notices: string[]; skills: string[] }> {
  const { kit, notices } = await readKit(folder)
  const target = kitDir(root, kit.id), files = await walk(folder)
  const patterns = [...kit.pieces.filter(p => ['docs', 'glossary', 'rules', 'assets'].includes(p.kind) && p.path).map(p => matcher(p.path!.replace(/\\/g, '/'))), /^madness-kit\.toml$/]
  await rm(target, { recursive: true, force: true })
  const hash = createHash('sha256')
  for (const file of files.sort()) {
    const rel = relative(folder, file).split(sep).join('/')
    if (!patterns.some(p => p.test(rel))) continue
    await mkdir(dirname(join(target, rel)), { recursive: true }); await cp(file, join(target, rel))
    hash.update(rel + '\0'); hash.update(await readFile(file))

  await writeLock(root, [...others, { id: kit.id, name: kit.name, version: kit.version, source: folder, hash: hash.digest('hex') }])
  return { kit, notices, skills: await rebuild(root) }
}

export async function deactivateKit(root: string, id: string): Promise<void> {
